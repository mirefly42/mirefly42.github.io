    320     #pixelsFromMemory(width, height, format, type, pixels_ptr) {
    321         const channels = this.#getChannelsCountFromFormat(format);
    322 
    323         if (pixels_ptr === 0) {
    324             return null;
    325         } else if (type === this.gl.UNSIGNED_BYTE) {
    326             return new Uint8Array(this.memory.buffer, pixels_ptr, width * height * channels);
    327         } else if (type === this.gl.UNSIGNED_SHORT_5_6_5 || type === UNSIGNED_SHORT_4_4_4_4 || type === UNSIGNED_SHORT_5_5_5_1) {
    328             throw new Error("not implemented");
