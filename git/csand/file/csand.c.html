    289 static bool csandKeyCallback(CsandKey key, CsandAction action, CsandModSet mods) {
    290     csandSendKeyStateToNuklear(key, action, mods);
    291 
    292     if (action != CSAND_ACTION_PRESS || any_nuklear_item_active) return false;
    293 
    294     switch (key) {
    295         case CSAND_KEY_0:
    296             draw_mat = MAT_AIR;
    297             return true;
    298         case CSAND_KEY_1:
    299             draw_mat = MAT_WALL;
    300             return true;
    301         case CSAND_KEY_2:
    302             draw_mat = MAT_SAND;
    303             return true;
    304         case CSAND_KEY_3:
    305             draw_mat = MAT_WATER;
    306             return true;
    307         case CSAND_KEY_4:
    308             draw_mat = MAT_FIRE_GAS;
    309             return true;
    310         case CSAND_KEY_5:
    311             draw_mat = MAT_WOOD;
    312             return true;
    313         case CSAND_KEY_6:
    314             draw_mat = MAT_COAL;
    315             return true;
    316         case CSAND_KEY_7:
    317             draw_mat = MAT_OIL;
    318             return true;
    319         case CSAND_KEY_8:
    320             draw_mat = MAT_HYDROGEN_GAS;
    321             return true;
    322         case CSAND_KEY_9:
    323             draw_mat = MAT_HYDROGEN_LIQUID;
    324             return true;
    325         case CSAND_KEY_SPACE:
    326             pause = !pause;
    327             return true;
    328         case CSAND_KEY_G:
    329             csandRendererSetGlow(!csandRendererGetGlow());
    330             return true;
    331         case CSAND_KEY_EQUAL:
    332             csandDoubleSimulationSpeed();
    333             return true;
    334         case CSAND_KEY_MINUS:
    335             csandHalveSimulationSpeed();
    336             return true;
    337         case CSAND_KEY_PERIOD:
    338             csandSimulateSingleFrame();
    339             return true;
    340         case CSAND_KEY_F:
    341             csandPlatformToggleFullscreen();
    342             return true;
    343         case CSAND_KEY_TAB:
    344             developer_menu_enabled = !developer_menu_enabled;
    345             return true;
    346         default:
    347             break;
    348     }
    349 
    350     return false;
    351 }

    506 static void csandSymmetricDraw(CsandVec2Us origin, unsigned short x, unsigned short y) {
    507     if (origin.y >= y) {
    508         if (origin.x >= x) {
    509             *csandGetMat(render_buf, origin.x - x, origin.y - y) = draw_mat;
    510         }
    511 
    512         if (origin.x + x < render_buf.width && x > 0) {
    513             *csandGetMat(render_buf, origin.x + x, origin.y - y) = draw_mat;
    514         }
    515     }
    516 
    517     if (origin.y + y < render_buf.height && y > 0) {
    518         if (origin.x >= x) {
    519             *csandGetMat(render_buf, origin.x - x, origin.y + y) = draw_mat;
    520         }
    521 
    522         if (origin.x + x < render_buf.width && x > 0) {
    523             *csandGetMat(render_buf, origin.x + x, origin.y + y) = draw_mat;
    524         }
    525     }
    526 }

    528 static void csandDraw(CsandVec2Us origin) {
    529     const unsigned short r = draw_diameter / 2;
    530     const unsigned long r_square = (unsigned long)draw_diameter*draw_diameter / (2*2);
    531 
    532     for (unsigned short y = 0; y <= r; ++y) {
    533         for (unsigned short x = 0; x <= r && (unsigned long)x*x + (unsigned long)y*y < r_square; ++x) {
    534             csandSymmetricDraw(origin, x, y);
    535         }
    536     }
    537 }

    539 static void csandRenderCallback(double time) {
    540     struct nk_context *nk_ctx = csandRendererNuklearContext();
    541     nk_input_end(nk_ctx);
    542 
    543     csandDrawButtons();
    544 
    545     if (developer_menu_enabled) {
    546         csandDrawDeveloperMenu();
    547     }
    548 
    549     any_nuklear_item_active = nk_item_is_any_active(nk_ctx);
    550 
    551     bool draw = !any_nuklear_item_active && csandPlatformIsMouseButtonPressed(CSAND_MOUSE_BUTTON_LEFT);
    552     CsandVec2Us cur_pos = csandRendererScreenSpaceToWorldSpace(csandPlatformGetCursorPos());
    553 
    554     if (time >= next_tick_time && (!pause || input_next_frame)) {
    555         input_next_frame = false;
    556         for (unsigned long i = 0; i < speed; i++) {
    557             csandSimulate(render_buf);
    558             if (draw) {
    559                 csandDraw(cur_pos);
    560             }
    561         }
    562         next_tick_time = time + TARGET_TICK_DELAY;
    563     } else if (draw) {
    564         csandDraw(cur_pos);
    565     }
    566 
    567     csandRendererRender(data, WIDTH, HEIGHT);
    568     nk_clear(nk_ctx);
    569 
    570     nk_input_begin(nk_ctx);
    571 }

    573 static void csandSimulate(CsandRenderBuffer buf) {
    574     for (unsigned int y = 0; y < buf.height; y++) {
    575         for (unsigned int x = 0; x < buf.width; x++) {
    576             unsigned char mat = *csandGetMat(buf, x, y);
    577 
    578             if (mat & MAT_UPDATED_BIT) {
    579                 continue;
    580             }
    581             CsandMaterialProperties mat_props = materials[mat];
    582 
    583             if (csandChance(mat_props.decay_prob)) {
    584                 *csandGetMat(buf, x, y) = mat_props.decay_mat | MAT_UPDATED_BIT;
    585                 continue;
    586             }
    587 
    588             int dx = csandRand() % 3 - 1;
    589             int dy = mat_props.kind == MAT_KIND_POWDER ? -1 : -(csandRand() & 1);
    590 
    591             if (!csandInBounds(buf, x + dx, y + dy)) {
    592                 continue;
    593             }
    594 
    595             unsigned int sx = x + dx;
    596             unsigned int sy = y + dy;
    597             unsigned char swap_mat = *csandGetMat(buf, sx, sy);
    598             if (swap_mat & MAT_UPDATED_BIT) {
    599                 continue;
    600             }
    601 
    602             CsandMaterialProperties swap_mat_props = materials[swap_mat];
    603 
    604             if (csandMatIsFire(mat)) {
    605                 tryIgnite(buf, sx, sy);
    606                 swap_mat = *csandGetMat(buf, sx, sy) & (~MAT_UPDATED_BIT);
    607                 swap_mat_props = materials[swap_mat];
    608             } else if (csandMatIsFire(swap_mat)) {
    609                 tryIgnite(buf, x, y);
    610                 mat = *csandGetMat(buf, x, y) & (~MAT_UPDATED_BIT);
    611                 mat_props = materials[mat];
    612             }
    613 
    614             if (mat_props.kind != MAT_KIND_SOLID && swap_mat_props.kind != MAT_KIND_SOLID && swap_mat_props.density < mat_props.density) {
    615                 *csandGetMat(buf, x, y) = swap_mat;
    616                 *csandGetMat(buf, sx, sy) = mat | MAT_UPDATED_BIT;
    617             }
    618         }
    619     }
    620 
    621     for (unsigned int i = 0; i < buf.width * buf.height; i++) {
    622         buf.data[i] &= ~MAT_UPDATED_BIT;
    623     }
    624 }

    626 static inline unsigned char *csandGetMat(CsandRenderBuffer buf, unsigned int x, unsigned int y) {
    627     return buf.data + buf.width * y + x;
    628 }
    629 
    630 static bool csandInBounds(CsandRenderBuffer buf, int x, int y) {
    631     return x >= 0 && x < buf.width && y >= 0 && y < buf.height;
    632 }

    634 static inline bool csandMatIsFire(unsigned char mat) {
    635     return mat == MAT_FIRE_GAS || mat == MAT_FIRE_POWDER || mat == MAT_FIRE_LIQUID;
    636 }
    637 
    638 static void tryIgnite(CsandRenderBuffer buf, unsigned int x, unsigned int y) {
    639     unsigned char mat = *csandGetMat(buf, x, y);
    640     CsandMaterialProperties mat_props = materials[mat];
    641 
    642     if (!csandChance(mat_props.ignition_prob)) {
    643         return;
    644     }
    645 
    646     const int air_range = 2;
    647     for (int dy = air_range; dy >= -air_range; dy--) {
    648         for (int dx = -air_range; dx <= air_range; dx++) {
    649             if (!(dx == 0 && dy == 0) && csandInBounds(buf, x + dx, y + dy) && *csandGetMat(buf, x + dx, y + dy) == MAT_AIR) {
    650                 switch (mat_props.kind) {
    651                     case MAT_KIND_SOLID:
    652                     case MAT_KIND_POWDER:
    653                         mat = csandRand() & 1 ? MAT_FIRE_GAS : MAT_FIRE_POWDER;
    654                         break;
    655                     case MAT_KIND_FLUID:
    656                         mat = csandRand() & 1 ? MAT_FIRE_GAS : MAT_FIRE_LIQUID;
    657                         break;
    658                     case MAT_KINDS_COUNT:
    659                         break;
    660                 }
    661 
    662                 *csandGetMat(buf, x, y) = mat | MAT_UPDATED_BIT;
    663                 return;
    664             }
    665         }
    666     }
