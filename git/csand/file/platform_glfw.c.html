     28 void csandPlatformToggleFullscreen(void) {
     29     if (glfwGetWindowMonitor(platform.window)) {
     30         glfwSetWindowMonitor(
     31             platform.window,
     32             NULL,
     33             platform.windowed_mode_pos.x,
     34             platform.windowed_mode_pos.y,
     35             platform.windowed_mode_size.x,
     36             platform.windowed_mode_size.y,
     37             GLFW_DONT_CARE
     38         );
     39     } else {
     40         glfwGetWindowPos(
     41             platform.window,
     42             &platform.windowed_mode_pos.x,
     43             &platform.windowed_mode_pos.y
     44         );
     45 
     46         glfwGetWindowSize(
     47             platform.window,
     48             &platform.windowed_mode_size.x,
     49             &platform.windowed_mode_size.y
     50         );

     52         GLFWmonitor *mon = glfwGetPrimaryMonitor();
     53         const GLFWvidmode *mode = glfwGetVideoMode(mon);
     54         glfwSetWindowMonitor(
     55             platform.window,
     56             mon,
     57             0, 0,
     58             mode->width, mode->height,
     59             GLFW_DONT_CARE
     60         );

     64 static void csandGlfwKeyCallback(GLFWwindow *window, int key, int scancode, int action, int mods) {
     65     (void)window;
     66     (void)scancode;
     67     (void)mods;
     68 
     69     if (!platform.key_callback) {
     70         return;
     71     }
     72 
     73     switch (action) {
     74         case GLFW_RELEASE:
     75         case GLFW_PRESS:
     76             platform.key_callback(key, action, mods & (GLFW_MOD_SHIFT | GLFW_MOD_CONTROL | GLFW_MOD_ALT));

    183 unsigned int csandPlatformIsMouseButtonPressed(CsandMouseButton button) {
    184     int glfw_button;
    185     switch (button) {
    186         case CSAND_MOUSE_BUTTON_LEFT:
    187             glfw_button = GLFW_MOUSE_BUTTON_LEFT;
    188             break;
    189         case CSAND_MOUSE_BUTTON_RIGHT:
    190             glfw_button = GLFW_MOUSE_BUTTON_RIGHT;
    191             break;
    192         default:
    193             return 0;
    194     }
    195 
    196     return glfwGetMouseButton(platform.window, glfw_button) == GLFW_PRESS;
