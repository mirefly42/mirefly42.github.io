    281     nk_convert(
    282         &csand_renderer.nk_ctx,
    283         &csand_renderer.cmds,

    297     GLint uv_attr_loc = glGetAttribLocation(csand_renderer.nuklear_program, "uv");
    298     glVertexAttribPointer(uv_attr_loc, 2, GL_FLOAT, GL_FALSE, sizeof(CsandNuklearVertex), (void *)offsetof(CsandNuklearVertex, uv));
    299     glEnableVertexAttribArray(uv_attr_loc);
    300

    303     glEnableVertexAttribArray(color_attr_loc);
    304 
    305     glUseProgram(csand_renderer.nuklear_program);
    306     setActiveTextureUnit(CSAND_TEXTURE_UNIT_FONT);
    307 
    308     glEnable(GL_BLEND);
    309     glBlendEquation(GL_FUNC_ADD);
    310     glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    311     glEnable(GL_SCISSOR_TEST);
    312 
    313     glViewport(0, 0, csand_renderer.framebuffer_size.x, csand_renderer.framebuffer_size.y);

    315 
    316     const struct nk_draw_command *cmd = NULL;

    318     nk_draw_foreach(cmd, &csand_renderer.nk_ctx, &csand_renderer.cmds) {
    319         if (cmd->elem_count == 0) {
    320             continue;
    321         }
    322 
    323         glScissor(cmd->clip_rect.x, csand_renderer.framebuffer_size.y - 1 - cmd->clip_rect.y - cmd->clip_rect.h, cmd->clip_rect.w, cmd->clip_rect.h);

    327 
    328     glDisable(GL_SCISSOR_TEST);
    329     glDisable(GL_BLEND);
    330 
    331     nk_buffer_clear(&csand_renderer.elements);
    332     nk_buffer_clear(&csand_renderer.vertices);
    333     nk_buffer_clear(&csand_renderer.cmds);
    334 }

    351 void csandRendererRender(unsigned char *data, unsigned short width, unsigned short height) {
    352     if (width != csand_renderer.world_size.x || height != csand_renderer.world_size.y) {
    353         csandUpdateWorldSize((CsandVec2Us){width, height});
    354     }
    355 
    356     glBindFramebuffer(GL_FRAMEBUFFER, csand_renderer.glow_fbo);
    357     glClearColor(0, 0, 0, 1);
    358     glClear(GL_COLOR_BUFFER_BIT);
    359 
    360     if (csand_renderer.glow_enabled) {
    361         glViewport(0, 0, width * CSAND_GLOW_RESOLUTION_FACTOR, height * CSAND_GLOW_RESOLUTION_FACTOR);
    362         drawFullscreenQuad(csand_renderer.glow_program);
    363     }
    364 
    365     glBindFramebuffer(GL_FRAMEBUFFER, 0);
    366 
    367     glClearColor(0.06, 0.12, 0.17, 1.0);
    368     glClear(GL_COLOR_BUFFER_BIT);
    369 
    370     setActiveTextureUnit(CSAND_TEXTURE_UNIT_RENDER);
    371     glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_LUMINANCE, GL_UNSIGNED_BYTE, data);
    372 
    373     glViewport(
    374         csand_renderer.viewport_offset.x,
    375         csand_renderer.viewport_offset.y,
    376         csand_renderer.viewport_size.x,
    377         csand_renderer.viewport_size.y
    378     );
    379 
    380     drawFullscreenQuad(csand_renderer.world_program);
    381 
    382     csandRendererRenderNuklear();
    383 }

    385 void csandRendererUpdateViewport(CsandVec2Us framebuffer_size) {
    386     csand_renderer.framebuffer_size = framebuffer_size;
    387     float scale_x = framebuffer_size.x / (float)csand_renderer.world_size.x;
    388     float scale_y = framebuffer_size.y / (float)csand_renderer.world_size.y;
    389 
    390     CsandVec2F viewport_size_float = csandVec2FMulScalar(
    391         CSAND_VEC2_CONVERT(CsandVec2F, csand_renderer.world_size),
    392         csandFMin(scale_x, scale_y)
    393     );
    394 
    395     csand_renderer.viewport_size = CSAND_VEC2_CONVERT(CsandVec2Us, viewport_size_float);
    396     csand_renderer.viewport_offset = csandVec2UsSub(
    397         csandVec2UsDivScalar(framebuffer_size, 2),
    398         csandVec2UsDivScalar(csand_renderer.viewport_size, 2)
    399     );
    400 }

    410 CsandVec2Us csandRendererScreenSpaceToWorldSpace(CsandVec2Us vec) {
    411     CsandVec2L result = csandVec2LClamp(
    412         csandVec2LDiv(
    413             csandVec2LMul(
    414                 csandVec2LSub(
    415                     (CsandVec2L){
    416                         vec.x,
    417                         csandPlatformGetWindowSize().y - 1 - vec.y,
    418                     },
    419                     CSAND_VEC2_CONVERT(CsandVec2L, csand_renderer.viewport_offset)
    420                 ),
    421                 CSAND_VEC2_CONVERT(CsandVec2L, csand_renderer.world_size)
    422             ),
    423             CSAND_VEC2_CONVERT(CsandVec2L, csand_renderer.viewport_size)
    424         ),
    425         (CsandVec2L){0},
    426         csandVec2LSubScalar(CSAND_VEC2_CONVERT(CsandVec2L, csand_renderer.world_size), 1)
    427     );
    428 
    429     return CSAND_VEC2_CONVERT(CsandVec2Us, result);

    436 static GLuint csandLoadShaderProgram(
    437     const char *program_name,
    438     const char *vertex_shader_name, const char *vertex_shader_src, size_t vertex_shader_length,
    439     const char *fragment_shader_name, const char *fragment_shader_src, size_t fragment_shader_length
    440 ) {
    441     GLuint vertex_shader = csandLoadShader(vertex_shader_name, vertex_shader_src, vertex_shader_length, GL_VERTEX_SHADER);
    442     GLuint fragment_shader = csandLoadShader(fragment_shader_name, fragment_shader_src, fragment_shader_length, GL_FRAGMENT_SHADER);
    443     GLuint program = glCreateProgram();
    444     glAttachShader(program, vertex_shader);
    445     glAttachShader(program, fragment_shader);
    446     glLinkProgram(program);
    447 
    448     GLint status;
    449     glGetProgramiv(program, GL_LINK_STATUS, &status);
    450 
    451     if (status != GL_TRUE) {
    452         char info_log[1024];
    453         glGetProgramInfoLog(program, sizeof(info_log), NULL, info_log);
    454         csandPlatformPrintErr("failed to link shader program '");
    455         csandPlatformPrintErr(program_name);
    456         csandPlatformPrintErr("': ");
    457         csandPlatformPrintErr(info_log);
    458         csandPlatformPrintErr("\n");
    459         glDeleteProgram(program);
    460         program = 0;
    461     }
    462 
    463     glDeleteShader(fragment_shader);
    464     glDeleteShader(vertex_shader);
    465 
    466     return program;
    467 }

    469 static GLuint csandLoadShader(const char *name, const char *src, size_t size, GLenum type) {
    470     GLenum shader = glCreateShader(type);

    472     GLint length[1] = {size};
    473     glShaderSource(shader, 1, (const char **)&src, length);
    474 
    475     glCompileShader(shader);
    476 
    477     GLint status;
    478     glGetShaderiv(shader, GL_COMPILE_STATUS, &status);
    479     if (status != GL_TRUE) {
    480         char info_log[1024];
    481 
    482         glGetShaderInfoLog(shader, sizeof(info_log), NULL, info_log);
    483         csandPlatformPrintErr("failed to compile shader '");
    484         csandPlatformPrintErr(name);
    485         csandPlatformPrintErr("': ");
    486         csandPlatformPrintErr(info_log);
    487         csandPlatformPrintErr("\n");
    488 
    489         glDeleteShader(shader);
    490         return 0;
    491     }
    492 
    493     return shader;
    494 }
