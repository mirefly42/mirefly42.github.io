+    static inline type csand##name##Clamp(type x, type min, type max) { \
+        if (x <= min) { \
+            return min; \
+        } else if (x <= max) { \
+            return x; \
+        } else { \
+            return max; \
+        } \
+    }
+
+#define CSAND_GEN_MIN(type, name) \
+    static inline type csand##name##Min(type a, type b) { \
+        return a <= b ? a : b; \
+    }
